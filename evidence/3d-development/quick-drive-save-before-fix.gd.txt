extends SceneTree
const Main = preload("res://scenes/main.tscn")
const Store = preload("res://scripts/save_store.gd")
const Content = preload("res://scripts/content.gd")
var game: Node
var base := "/tmp/garage-final-reward-review-"+str(OS.get_process_id())
func _initialize() -> void:
	call_deferred("run")
func labels(node: Node) -> String:
	var text := str(node.text)+"\n" if node is Label or node is Button else ""
	for child in node.get_children():
		text += labels(child)
	return text
func run() -> void:
	game=Main.instantiate()
	game.store=Store.new(base)
	root.add_child(game)
	await process_frame
	game.set_physics_process(false)
	game.start_course(Content.COURSES[0])
	DirAccess.make_dir_absolute(base+".tmp")
	var event:=InputEventKey.new()
	event.physical_keycode=KEY_RIGHT
	event.pressed=true
	game.router._input(event)
	for frame in range(60*180):
		game._physics_process(1.0/60.0)
		if game.run_finished:
			break
	print("REPRO finished=",game.run_finished," gp=",game.state.gp," last_settled=",game.store.data.economy.last_settled," run_id=",game.active_run_id," observed=",game.economy._observation)
	print("REPRO visible UI:\n",labels(game.ui))
	DirAccess.remove_absolute(base+".tmp")
	game._finish()
	print("REPRO after storage unblocked and finish retried gp=",game.store.data.gp," run_finished=",game.run_finished)
	game.sound.stop_all()
	game.queue_free()
	await create_timer(0.3).timeout
	for suffix in [".json",".bak",".tmp"]:
		DirAccess.remove_absolute(base+suffix)
	quit()
