extends SceneTree
const Main = preload("res://scenes/main.tscn")
const Store = preload("res://scripts/save_store.gd")
const Content = preload("res://scripts/content.gd")
var base := "/tmp/garage-final-begin-review-"+str(OS.get_process_id())
func _initialize() -> void:
	call_deferred("run")
func labels(node: Node) -> String:
	var text := str(node.text)+"\n" if node is Label or node is Button else ""
	for child in node.get_children():
		text += labels(child)
	return text
func run() -> void:
	var game=Main.instantiate()
	game.store=Store.new(base)
	root.add_child(game)
	await process_frame
	game.set_physics_process(false)
	DirAccess.make_dir_absolute(base+".tmp")
	game.start_course(Content.COURSES[0])
	print("REPRO page=",game.page," run_id=",game.active_run_id," router_active=",game.router.active," economy_warning=",game.economy.warning)
	var visible:=labels(game.ui)
	print("REPRO warning_visible=",visible.contains(game.economy.warning),"\nREPRO visible UI:\n",visible)
	DirAccess.remove_absolute(base+".tmp")
	game.sound.stop_all()
	game.queue_free()
	await create_timer(0.3).timeout
	for suffix in [".json",".bak",".tmp"]:
		DirAccess.remove_absolute(base+suffix)
	quit()
